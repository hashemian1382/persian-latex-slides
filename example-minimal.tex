\documentclass[11pt,aspectratio=169]{beamer}

\makeatletter
\def\input@path{{./}{theme/}{config/}{content/}{universities/}}

  progressbar = foot,
  numbering   = fraction,
  sectionpage = false,      % ارائهٔ کوتاه، اسلاید جداکننده لازم نیست
}

\begin{document}

\PTtitleframe

\section{نمونه}

\begin{frame}{یک اسلاید نمونه}{زیرعنوان اختیاری}
  \begin{itemize}
    \item فهرست‌ها، \term{اصطلاح‌ها} و \hl{تأکیدها} همه آماده‌اند.
    \item متن انگلیسی درون \en{...} می‌آید: \en{Retrieval-Augmented Generation}.
  \end{itemize}

  \begin{PTdef}[\en{RAG}]
    الگویی که پیش از تولید پاسخ، اسناد مرتبط را از منبعی بیرونی بازیابی می‌کند.
  \end{PTdef}

  \begin{PTnote}
    جعبه‌های آمادهٔ دیگر: \en{PTthm}، \en{PTpros}، \en{PTcons}، \en{PTresult}،
    \en{PTexample}، \en{PTcallout}.
  \end{PTnote}
\end{frame}

\PTclosingframe

\end{document}
